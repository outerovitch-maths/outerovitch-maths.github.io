\begin{exercicedevoir}[5][Arrondis]
\begin{exempletraite}
\textbf{Arrondir $0{,}25$ au dixième.} Chiffre des dixièmes : $2$
($0{,}\underline{2}5$). Chiffre \textbf{juste après} : $5$.
\begin{itemize}[nosep]
\item de 0 à 4 : on garde le chiffre ; \quad de 5 à 9 : on ajoute 1.
\item Ici $5$, donc $2$ devient $3$ : \textbf{$0{,}25 \approx 0{,}3$}.
\end{itemize}
Autre exemple : $3{,}141$ au centième : le chiffre suivant est $1$, on
garde : $3{,}14$.
\end{exempletraite}

\renewcommand{\arraystretch}{1.2}
\begin{center}
\begin{tabular}{|c|c|p{6.5cm}|c|}
\hline
\textbf{Nombre} & \textbf{Arrondi au\ldots} & \textbf{Explication} &
\textbf{Arrondi} \\ \hline
$0{,}25$ & dixième & \textit{chiffre suivant : 5, j'ajoute 1} &
$0{,}3$ \\ \hline
$0{,}73$ & dixième & \rep & \\ \hline
$4{,}86$ & dixième & \rep & \\ \hline
$12{,}348$ & centième & \rep & \\ \hline
$7{,}6521$ & centième & \rep & \\ \hline
$2{,}4136$ & millième & \rep & \\ \hline
$3{,}7058$ & millième & \rep & \\ \hline
\end{tabular}
\end{center}
\end{exercicedevoir}

\begin{exercicedevoir}[4][Arrondir une racine carrée]
\begin{exempletraite}
\textbf{Arrondir $\sqrt{30}$ au centième.} \\
1. Calculatrice, avec plusieurs chiffres après la virgule :
$\sqrt{30} = 5{,}4772\ldots$ \\
2. Le chiffre après les centièmes est $7$, j'ajoute 1 :
\textbf{$\sqrt{30} \approx 5{,}48$}.
\end{exempletraite}

\renewcommand{\arraystretch}{1.2}
\begin{center}
\begin{tabular}{|c|c|p{4.5cm}|p{3cm}|}
\hline
\textbf{Racine} & \textbf{Arrondi au\ldots} &
\textbf{Calculatrice} & \textbf{Arrondi} \\ \hline
$\sqrt{30}$ & centième & $5{,}4772\ldots$ & $5{,}48$ \\ \hline
$\sqrt{2}$ & dixième & \rep & \\ \hline
$\sqrt{50}$ & dixième & \rep & \\ \hline
$\sqrt{10}$ & centième & \rep & \\ \hline
$\sqrt{45}$ & centième & \rep & \\ \hline
$\sqrt{7}$ & millième & \rep & \\ \hline
\end{tabular}
\end{center}
\end{exercicedevoir}
