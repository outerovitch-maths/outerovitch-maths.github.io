\newif\ifcorrige
% Réponse dans une case de largeur fixe : \rep[largeur]{réponse}
% (pointillés sur le sujet, réponse en rouge sur le corrigé, même place)
\newcommand{\rep}[2][0.8cm]{\makebox[#1]{\ifcorrige\textcolor{red}{$#2$}\else\dotfill\fi}}
% Facteur commun à barrer : imprimé sur le sujet, barré en rouge sur le corrigé
\newcommand{\barre}[1]{\ifcorrige{\renewcommand{\CancelColor}{\color{red}}\cancel{#1}}\else #1\fi}
% Item déjà fait (exemple) : fond grisé
\newcommand{\fait}[1]{\colorbox{gray!15}{#1}}

\begin{tcolorbox}[width=\linewidth-1cm,colback=white,colframe=black,boxrule=0.6pt,arc=2mm,left=2mm,right=2mm,top=0.5mm,bottom=0.5mm]
\textbf{Méthode :} on écrit le haut \textbf{et} le bas comme un produit
avec le \textbf{même facteur},\newline puis on \textbf{barre} ce facteur
en haut et en bas.
\hfill $\dfrac{6}{8}=\dfrac{\cancel{2} \times 3}{\cancel{2} \times 4}=\dfrac{3}{4}$\hspace*{1cm}
\end{tcolorbox}
\vspace{-1.5em}
\begin{exercice}[1][Simplifie par 2]
Complète, puis barre le $2$ en haut et en bas. \hfill \marqueex\ = exemple

\renewcommand{\arraystretch}{2.25}
\begin{tabularx}{\linewidth}{@{}XXX@{}}
\fait{a. \ $\dfrac{6}{8}=\dfrac{\cancel{2} \times 3}{\cancel{2} \times 4}=\dfrac{3}{4}$} &
b. \ $\dfrac{10}{14}=\dfrac{\barre{2} \times \rep{5}}{\barre{2} \times \rep{7}}=\dfrac{\rep{5}}{\rep{7}}$ &
c. \ $\dfrac{4}{18}=\dfrac{\barre{2} \times \rep{2}}{\barre{2} \times \rep{9}}=\dfrac{\rep{2}}{\rep{9}}$ \\
d. \ $\dfrac{12}{14}=\dfrac{\barre{2} \times \rep{6}}{\barre{2} \times \rep{7}}=\dfrac{\rep{6}}{\rep{7}}$ &
e. \ $\dfrac{16}{18}=\dfrac{\barre{2} \times \rep{8}}{\barre{2} \times \rep{9}}=\dfrac{\rep{8}}{\rep{9}}$ &
f. \ $\dfrac{8}{10}=\dfrac{\barre{2} \times \rep{4}}{\barre{2} \times \rep{5}}=\dfrac{\rep{4}}{\rep{5}}$ \\
\end{tabularx}
\end{exercice}
