% Réponse en rouge, ou pointillés de même largeur : \rep[largeur]{réponse}
\newcommand{\rep}[2][0.9cm]{\ifcorrige\makebox[#1]{\vphantom{$0$}\textcolor{red}{#2}}\else\makebox[#1]{\vphantom{$0$}\dotfill}\fi}
% Réponse de l'exemple : toujours écrite, en noir, à la même place
\newcommand{\ex}[2][0.9cm]{\makebox[#1]{\vphantom{$0$}#2}}
% Fraction à compléter : \repfrac{num}{den} ; fraction de l'exemple : \exfrac{num}{den}
\newcommand{\repfrac}[2]{\dfrac{\rep{$#1$}}{\rep{$#2$}}}
\newcommand{\exfrac}[2]{\dfrac{\ex{$#1$}}{\ex{$#2$}}}
% Bande de 4,4 cm partagée en #1 parts égales, dont #2 coloriées (décalée de #3 vers le bas)
\newcommand{\bandepic}[3]{%
  \pgfmathsetmacro{\lp}{4.4/#1}
  \fill[gray!45] (0,-#3) rectangle ({#2*\lp},-#3+0.5);
  \foreach \i in {1,...,#1}{\draw ({(\i-1)*\lp},-#3) rectangle ({\i*\lp},-#3+0.5);}
  \draw[thick] (0,-#3) rectangle (4.4,-#3+0.5);}
% Paire de bandes : \paire{parts1}{col1}{parts2}{col2}
\newcommand{\paire}[4]{%
\begin{tikzpicture}[baseline=-0.25cm]
  \bandepic{#1}{#2}{0}
  \bandepic{#3}{#4}{0.75}
\end{tikzpicture}}
% Mot entouré en rouge dans le corrigé (même place dans le sujet)
\newcommand{\pasentoure}[1]{\tikz[baseline=(n.base)]\node[thick, ellipse, inner sep=1.5pt] (n) {#1};}
\newcommand{\entoureex}[1]{\tikz[baseline=(n.base)]\node[draw, thick, ellipse, inner sep=1.5pt] (n) {#1};}
\newcommand{\entoure}[1]{\ifcorrige\tikz[baseline=(n.base)]\node[draw=red, thick, ellipse, inner sep=1.5pt] (n) {#1};\else\pasentoure{#1}\fi}

\begin{tcolorbox}[colback=white, colframe=black, boxrule=0.8pt, arc=2mm, left=2mm, right=2mm, top=1mm, bottom=1mm]
\begin{minipage}{0.74\linewidth}\textbf{À retenir :} deux fractions sont \textbf{égales} si on multiplie (ou divise)
le haut \textbf{ET} le bas par le \textbf{même nombre}.\end{minipage}
\hfill $\dfrac{2}{3} = \dfrac{2 \times 4}{3 \times 4} = \dfrac{8}{12}$
\end{tcolorbox}

\begin{exercice}[1][Écris la fraction coloriée]
Les bandes ont la même longueur. Écris la fraction coloriée de chaque bande.

\medskip
\renewcommand{\arraystretch}{1}
\begin{tabular}{@{}l@{\hspace{2mm}}c@{\hspace{3mm}}c@{\hspace{5mm}}l@{\hspace{2mm}}c@{\hspace{3mm}}c@{}}
\textbf{Exemple :} & \paire{2}{1}{4}{2} & $\exfrac{1}{2} = \exfrac{2}{4}$ &
\textbf{b)} & \paire{3}{1}{6}{2} & $\repfrac{1}{3} = \repfrac{2}{6}$ \\[0.6cm]
\textbf{c)} & \paire{4}{3}{8}{6} & $\repfrac{3}{4} = \repfrac{6}{8}$ &
\textbf{d)} & \paire{5}{2}{10}{4} & $\repfrac{2}{5} = \repfrac{4}{10}$ \\
\end{tabular}

\end{exercice}

\vspace{-0.8em}
\begin{exercice}[1][Complète]
On multiplie le haut et le bas par le nombre donné. Complète.

\renewcommand{\arraystretch}{2.3}
\begin{tabularx}{\linewidth}{@{}XXX@{}}
\textbf{Exemple :} $\dfrac{1}{2} = \dfrac{1 \times 2}{2 \times 2} = \exfrac{2}{4}$ &
\textbf{b)} $\dfrac{3}{4} = \dfrac{3 \times 2}{4 \times 2} = \repfrac{6}{8}$ &
\textbf{c)} $\dfrac{1}{3} = \dfrac{1 \times 3}{3 \times 3} = \repfrac{3}{9}$ \\
\textbf{d)} $\dfrac{2}{5} = \dfrac{2 \times 3}{5 \times 3} = \repfrac{6}{15}$ &
\textbf{e)} $\dfrac{1}{4} = \dfrac{1 \times 5}{4 \times 5} = \repfrac{5}{20}$ &
\textbf{f)} $\dfrac{3}{7} = \dfrac{3 \times 10}{7 \times 10} = \repfrac{30}{70}$ \\
\end{tabularx}
\end{exercice}

\vspace{-0.8em}
\begin{exercice}[2][Par combien a-t-on multiplié ?]
Les fractions sont égales. Écris le nombre par lequel on a multiplié en haut et en bas.

\renewcommand{\arraystretch}{2.3}
\begin{tabularx}{\linewidth}{@{}XXX@{}}
\textbf{Exemple :} $\dfrac{1}{3} = \dfrac{1 \times \ex[0.7cm]{$2$}}{3 \times \ex[0.7cm]{$2$}} = \dfrac{2}{6}$ &
\textbf{b)} $\dfrac{3}{4} = \dfrac{3 \times \rep[0.7cm]{$3$}}{4 \times \rep[0.7cm]{$3$}} = \dfrac{9}{12}$ &
\textbf{c)} $\dfrac{2}{5} = \dfrac{2 \times \rep[0.7cm]{$5$}}{5 \times \rep[0.7cm]{$5$}} = \dfrac{10}{25}$ \\
\textbf{d)} $\dfrac{1}{2} = \dfrac{1 \times \rep[0.7cm]{$10$}}{2 \times \rep[0.7cm]{$10$}} = \dfrac{10}{20}$ &
\textbf{e)} $\dfrac{2}{3} = \dfrac{2 \times \rep[0.7cm]{$4$}}{3 \times \rep[0.7cm]{$4$}} = \dfrac{8}{12}$ & \\
\end{tabularx}
\end{exercice}

\vspace{-0.8em}
\begin{exercice}[2][Égales ou pas ?]
Complète. Si c'est le même nombre en haut et en bas, entoure «~oui~», sinon «~non~».

\medskip
\renewcommand{\arraystretch}{1}
\renewcommand{\tabularxcolumn}[1]{m{##1}}
\begin{tabularx}{\linewidth}{|>{\centering\arraybackslash\rule[-4mm]{0pt}{10mm}}m{4.2cm}|>{\centering\arraybackslash}X|>{\centering\arraybackslash}X|>{\centering\arraybackslash}m{3.4cm}|}
\hline
\multicolumn{1}{|c|}{\rule[-2.5mm]{0pt}{7mm}\textbf{Les deux fractions}} & \textbf{En haut} & \textbf{En bas} & \textbf{Égales ?} \\ \hline
\textbf{Exemple :} \quad $\dfrac{2}{3}$ et $\dfrac{4}{6}$ & $2 \times \ex[0.8cm]{$2$} = 4$ & $3 \times \ex[0.8cm]{$2$} = 6$ & \entoureex{oui} \qquad \pasentoure{non} \\ \hline
$\dfrac{1}{4}$ et $\dfrac{3}{12}$ & $1 \times \rep[0.8cm]{$3$} = 3$ & $4 \times \rep[0.8cm]{$3$} = 12$ & \entoure{oui} \qquad \pasentoure{non} \\ \hline
$\dfrac{3}{5}$ et $\dfrac{6}{15}$ & $3 \times \rep[0.8cm]{$2$} = 6$ & $5 \times \rep[0.8cm]{$3$} = 15$ & \pasentoure{oui} \qquad \entoure{non} \\ \hline
$\dfrac{4}{5}$ et $\dfrac{20}{25}$ & $4 \times \rep[0.8cm]{$5$} = 20$ & $5 \times \rep[0.8cm]{$5$} = 25$ & \entoure{oui} \qquad \pasentoure{non} \\ \hline
$\dfrac{3}{10}$ et $\dfrac{30}{100}$ & $3 \times \rep[0.8cm]{$10$} = 30$ & $10 \times \rep[0.8cm]{$10$} = 100$ & \entoure{oui} \qquad \pasentoure{non} \\ \hline
\end{tabularx}
\end{exercice}
