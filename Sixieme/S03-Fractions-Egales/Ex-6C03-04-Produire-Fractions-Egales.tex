\begin{tcolorbox}[colback=gray!8, colframe=black, boxrule=0.6pt, left=2mm, right=2mm, top=1mm, bottom=1mm]
\textbf{Rappel.} On multiplie (ou on divise) le haut \textbf{et} le bas par le \textbf{même} nombre. \hfill Exemple : $\dfrac{2}{3}=\dfrac{2\times 4}{3\times 4}=\dfrac{8}{12}$
\end{tcolorbox}
\vspace{-0.6em}

\begin{exercice}[1][Je multiplie]
Complète. L'exemple grisé est déjà fait.

\textbf{1.} Multiplie le haut et le bas par $2$.

\medskip
\hfill \modele{$\dfrac{2}{3}=\dfrac{2\times 2}{3\times 2}=\dfrac{\vu{4}}{\vu{6}}$} \hfill
$\dfrac{1}{4}=\dfrac{1\times 2}{4\times 2}=\dfrac{\rep{2}}{\rep{8}}$ \hfill
$\dfrac{3}{5}=\dfrac{3\times 2}{5\times 2}=\dfrac{\rep{6}}{\rep{10}}$ \hfill
$\dfrac{5}{6}=\dfrac{5\times 2}{6\times 2}=\dfrac{\rep{10}}{\rep{12}}$ \hfill
$\dfrac{4}{7}=\dfrac{4\times 2}{7\times 2}=\dfrac{\rep{8}}{\rep{14}}$ \hfill\null

\medskip
\textbf{2.} Multiplie le haut et le bas par $10$.

\medskip
\hfill \modele{$\dfrac{1}{2}=\dfrac{1\times 10}{2\times 10}=\dfrac{\vu{10}}{\vu{20}}$} \hfill
$\dfrac{3}{4}=\dfrac{3\times 10}{4\times 10}=\dfrac{\rep{30}}{\rep{40}}$ \hfill
$\dfrac{2}{5}=\dfrac{2\times 10}{5\times 10}=\dfrac{\rep{20}}{\rep{50}}$ \hfill
$\dfrac{7}{10}=\dfrac{7\times 10}{10\times 10}=\dfrac{\rep{70}}{\rep{100}}$ \hfill
$\dfrac{4}{9}=\dfrac{4\times 10}{9\times 10}=\dfrac{\rep{40}}{\rep{90}}$ \hfill\null
\end{exercice}

\vspace{-0.8em}
\begin{exercice}[1][Je divise]
Divise le haut et le bas par le nombre donné. Complète.

\medskip
\hfill \modele{$\dfrac{6}{8}=\dfrac{6\div 2}{8\div 2}=\dfrac{\vu{3}}{\vu{4}}$} \hfill
$\dfrac{12}{15}=\dfrac{12\div 3}{15\div 3}=\dfrac{\rep{4}}{\rep{5}}$ \hfill
$\dfrac{10}{15}=\dfrac{10\div 5}{15\div 5}=\dfrac{\rep{2}}{\rep{3}}$ \hfill
$\dfrac{30}{50}=\dfrac{30\div 10}{50\div 10}=\dfrac{\rep{3}}{\rep{5}}$ \hfill
$\dfrac{8}{20}=\dfrac{8\div 4}{20\div 4}=\dfrac{\rep{2}}{\rep{5}}$ \hfill\null
\end{exercice}

\vspace{-0.8em}
\begin{exercice}[2][Je trouve le nombre qui manque]
Trouve par combien on multiplie, puis complète.

\textbf{1.} Le nombre qui manque est en haut.

\medskip
\hfill \modele{\begin{tabular}{@{}l@{}}$2\times\vu[0.6cm]{3}=6$ \ donc \ $\times\,\vu[0.6cm]{3}$\\[0.4em]$\dfrac{1}{2}=\dfrac{1\times\vu[0.6cm]{3}}{2\times\vu[0.6cm]{3}}=\dfrac{\vu{3}}{6}$\end{tabular}} \hfill
\begin{tabular}{@{}l@{}}$3\times\rep[0.6cm]{4}=12$ \ donc \ $\times\,\rep[0.6cm]{4}$\\[0.4em]$\dfrac{1}{3}=\dfrac{1\times\rep[0.6cm]{4}}{3\times\rep[0.6cm]{4}}=\dfrac{\rep{4}}{12}$\end{tabular} \hfill
\begin{tabular}{@{}l@{}}$5\times\rep[0.6cm]{2}=10$ \ donc \ $\times\,\rep[0.6cm]{2}$\\[0.4em]$\dfrac{2}{5}=\dfrac{2\times\rep[0.6cm]{2}}{5\times\rep[0.6cm]{2}}=\dfrac{\rep{4}}{10}$\end{tabular} \hfill
\begin{tabular}{@{}l@{}}$4\times\rep[0.6cm]{10}=40$ \ donc \ $\times\,\rep[0.6cm]{10}$\\[0.4em]$\dfrac{3}{4}=\dfrac{3\times\rep[0.6cm]{10}}{4\times\rep[0.6cm]{10}}=\dfrac{\rep{30}}{40}$\end{tabular} \hfill\null

\medskip
\textbf{2.} Le nombre qui manque est en bas.

\medskip
\hfill \modele{\begin{tabular}{@{}l@{}}$1\times\vu[0.6cm]{2}=2$ \ donc \ $\times\,\vu[0.6cm]{2}$\\[0.4em]$\dfrac{1}{4}=\dfrac{1\times\vu[0.6cm]{2}}{4\times\vu[0.6cm]{2}}=\dfrac{2}{\vu{8}}$\end{tabular}} \hfill
\begin{tabular}{@{}l@{}}$2\times\rep[0.6cm]{5}=10$ \ donc \ $\times\,\rep[0.6cm]{5}$\\[0.4em]$\dfrac{2}{3}=\dfrac{2\times\rep[0.6cm]{5}}{3\times\rep[0.6cm]{5}}=\dfrac{10}{\rep{15}}$\end{tabular} \hfill
\begin{tabular}{@{}l@{}}$3\times\rep[0.6cm]{3}=9$ \ donc \ $\times\,\rep[0.6cm]{3}$\\[0.4em]$\dfrac{3}{5}=\dfrac{3\times\rep[0.6cm]{3}}{5\times\rep[0.6cm]{3}}=\dfrac{9}{\rep{15}}$\end{tabular} \hfill
\begin{tabular}{@{}l@{}}$1\times\rep[0.6cm]{4}=4$ \ donc \ $\times\,\rep[0.6cm]{4}$\\[0.4em]$\dfrac{1}{2}=\dfrac{1\times\rep[0.6cm]{4}}{2\times\rep[0.6cm]{4}}=\dfrac{4}{\rep{8}}$\end{tabular} \hfill\null
\end{exercice}
