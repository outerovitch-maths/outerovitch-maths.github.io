% Triangle XYZ dont les côtés sont prolongés (droites fines) ;
% dans le corrigé, les côtés de l'angle de sommet #1 sont repassés en rouge.
% #7, #8, #9 : direction (en degrés) de chaque étiquette par rapport à son sommet.
% \triangleCotes{X}{Y}{Z}{x_X,y_X}{x_Y,y_Y}{x_Z,y_Z}{dir_X}{dir_Y}{dir_Z}
\newcommand{\triangleCotes}[9]{%
  \begin{tikzpicture}[scale=0.62]
    \coordinate (P1) at (#4); \coordinate (P2) at (#5); \coordinate (P3) at (#6);
    \draw[gray] ($(P1)!-0.35!(P2)$) -- ($(P2)!-0.35!(P1)$);
    \draw[gray] ($(P2)!-0.35!(P3)$) -- ($(P3)!-0.35!(P2)$);
    \draw[gray] ($(P3)!-0.35!(P1)$) -- ($(P1)!-0.35!(P3)$);
    \draw[thick] (P1) -- (P2) -- (P3) -- cycle;
    \ifcorrige
      \draw[red, line width=1.6pt] ($(P2)!-0.35!(P1)$) -- (P1) -- ($(P3)!-0.35!(P1)$);
    \fi
    \node at ($(P1)+(#7:3.5mm)$) {$#1$};
    \node at ($(P2)+(#8:3.5mm)$) {$#2$};
    \node at ($(P3)+(#9:3.5mm)$) {$#3$};
  \end{tikzpicture}%
}

% Trois demi-droites de même origine O passant par A, B, C ;
% dans le corrigé, les côtés de l'angle AOC sont repassés en rouge.
\newcommand{\eventail}{%

    \coordinate (O) at (0,0);

    \draw[thick] (O) -- (5:3.6); \draw[thick] (O) -- (60:3.1); \draw[thick] (O) -- (140:3.1);

    \node at ($(A)+(-85:3mm)$) {$A$};
    \node at ($(B)+(-30:3mm)$) {$B$};
    \node at ($(C)+(230:3mm)$) {$C$};
    \node at (-90:3.5mm) {$O$};
  \end{tikzpicture}%

\begin{tcolorbox}[width=\linewidth-1cm,colback=white,colframe=black,boxrule=0.6pt,arc=2mm,left=2mm,right=2mm,top=0.5mm,bottom=0.5mm]
\begin{minipage}{0.64\linewidth}
\textbf{Rappel :} dans l'angle $\ang{ABC}$,
\begin{itemize}[nosep,leftmargin=1.2em]
\item le \textbf{sommet} est la lettre \textbf{du milieu} : le point $B$ ;
\item les \textbf{côtés} sont les \textbf{demi-droites} $[BA)$ et $[BC)$ :\newline
elles \textbf{partent du sommet} $B$ et ne s'arrêtent pas.
\end{itemize}
\smallskip
On écrit $[BA)$ : le crochet est du côté du sommet $B$,\newline
la parenthèse du côté où la demi-droite continue.
\end{minipage}\hfill
\begin{minipage}{0.34\linewidth}
\centering
\begin{tikzpicture}[scale=0.9]
  \coordinate (V) at (0,0);
  \coordinate (P) at (55:1.5); \coordinate (Q) at (5:1.7);
  \draw[gray!70!black, fill=gray!30] (0,0) -- (5:4.5mm) arc[start angle=5, delta angle=50, radius=4.5mm] -- cycle;
  \draw[thick] (55:2.4) -- (V) -- (5:2.9);
  \foreach \p in {P,Q} \draw (\p) ++(-2pt,-2pt) -- ++(4pt,4pt) ++(-4pt,0) -- ++(4pt,-4pt);
  \node at ($(P)+(145:3mm)$) {$A$};
  \node at ($(Q)+(-85:3mm)$) {$C$};
  \node at (210:3.5mm) {$B$};
  \node[font=\footnotesize, align=center] (s) at (-1.1,0.9) {sommet};
  \draw[->, >=stealth] (s) -- (-0.1,0.1);
  \node[font=\footnotesize, anchor=south east] at (55:2.4) {côté $[BA)$};
  \node[font=\footnotesize, anchor=south] at (5:2.5) {côté $[BC)$};
\end{tikzpicture}
\end{minipage}
\end{tcolorbox}
