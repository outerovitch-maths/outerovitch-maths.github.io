\textbf{Exercice 2 : Points et figures} \\
Complète le tableau avec $\in$ ou $\notin$. Un point appartient à une figure lorsqu'il est placé \emph{sur son tracé}.

\begin{center}
\begin{tikzpicture}
  % Triangle KLM, B sur [LM]
  \coordinate (K) at (0,0);
  \coordinate (L) at (3.5,0.3);
  \coordinate (M) at (1.2,3.5);
  \coordinate (B) at ($(L)!0.45!(M)$);
  \draw[thick] (K) -- (L) -- (M) -- cycle;

  % Carré RSTU, E à l'intérieur
  \coordinate (R) at (5,0);
  \coordinate (S) at (8,0);
  \coordinate (T) at (8,3);
  \coordinate (U) at (5,3);
  \coordinate (E) at (6.3,1.6);
  \draw[thick] (R) -- (S) -- (T) -- (U) -- cycle;

  % Cercle de centre O, rayon 2,5 : coupe [ST] en D (8 ; 2,2)
  \coordinate (O) at (10.4,1.5);
  \coordinate (A) at ($(O)+(60:2.5)$);
  \coordinate (C) at (11.3,2.4);
  \coordinate (D) at (8,2.2);
  \draw[thick] (O) circle (2.5);
  \node at ($(O)+(-40:2.95)$) {$\mathscr{C}$};

  \tkzDrawPoints[shape=cross out,size=9](K,L,M,R,S,T,U,O,A,B,C,D,E)
  \tkzLabelPoints[below left=0.5mm](K,R)
  \tkzLabelPoints[below right=0.5mm](L,S,O)
  \tkzLabelPoints[right=1mm](C)
  \tkzLabelPoints[above=1mm](M,U,E)
  \tkzLabelPoints[above left=0.5mm](T,D)
  \tkzLabelPoints[above right=0.5mm](A,B)
\end{tikzpicture}
\end{center}
